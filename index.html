; Definición de función que emula la lógica de estado trinario
define i32 @estado_trinario(i32 %nivel) {
entry:
  %es_positivo = icmp sgt i32 %nivel, 0
  br i1 %es_positivo, label %positivo, label %negativo

positivo:
  ret i32 1

negativo:
  ret i32 -1
}
